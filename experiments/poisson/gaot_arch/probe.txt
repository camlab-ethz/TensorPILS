# Stage 0 -- does every GAOT arm construct, how long is an epoch, and does it fit?
#
#     clsubmit experiments/poisson/gaot_arch/probe.txt
#
# Five epochs at the FULL production size (64^2, n_train=1024, batch 32), so the epoch time and
# the peak memory read off this log are the ones the 500-epoch sweep will actually see. Three
# jobs rather than one, because the three arms differ in what they build: `pls` assembles the
# multigrid V-cycle, `galerkin` runs the node-space residual, `data` stays on the grid.
#
# Read from the log:  the tqdm rate (epoch/s) -> walltime for 500 epochs;
#                     "torch.cuda.OutOfMemoryError" -> halve --batch_size or raise --gaot_patch.
# Results go to a throwaway directory: 5 epochs is not a result.

#!time 00:25:00
#!profile gpu1

# name                profile  command
gaot-probe-data       -        python -m tensorpils.cli --pde poisson --model gaot --loss data --grid_resolution 64 --n_train 1024 --n_val 128 --n_test 256 -k 4 --epochs 5 --batch_size 32 --seed 42 --optimizer adam --lr 1e-3 --lr_min 1e-6 --device cuda --no_checkpoint --output_dir output/poisson/gaot_arch/probe
gaot-probe-galerkin   -        python -m tensorpils.cli --pde poisson --model gaot --loss galerkin --grid_resolution 64 --n_train 1024 --n_val 128 --n_test 256 -k 4 --epochs 5 --batch_size 32 --seed 42 --optimizer adam --lr 1e-3 --lr_min 1e-6 --device cuda --no_checkpoint --output_dir output/poisson/gaot_arch/probe
gaot-probe-pls        -        python -m tensorpils.cli --pde poisson --model gaot --loss pls --grid_resolution 64 --n_train 1024 --n_val 128 --n_test 256 -k 4 --epochs 5 --batch_size 32 --seed 42 --optimizer adam --lr 1e-3 --lr_min 1e-6 --device cuda --no_checkpoint --output_dir output/poisson/gaot_arch/probe
